\chapter*{DANH MỤC CÁC TỪ VIẾT TẮT VÀ THUẬT NGỮ}
\addcontentsline{toc}{chapter}{Danh mục các từ viết tắt và thuật ngữ}

\begin{xltabular}{\textwidth}{l l L}
\toprule
\textbf{Ký hiệu} & \textbf{Thuật ngữ tiếng Anh} & \textbf{Ý nghĩa tiếng Việt} \\
\midrule
\endhead

\textbf{AP} & Access Point & Điểm truy cập không dây \\
\textbf{ARP} & Address Resolution Protocol & Giao thức phân giải địa chỉ (IP sang MAC) \\
\textbf{CA} & Certificate Authority & Tổ chức phát hành chứng chỉ số \\
\textbf{DAI} & Dynamic ARP Inspection & Kiểm tra gói tin ARP động trên thiết bị chuyển mạch \\
\textbf{DHCP} & Dynamic Host Configuration Protocol & Giao thức cấu hình máy chủ động \\
\textbf{DNS} & Domain Name System & Hệ thống phân giải tên miền \\
\textbf{DoS} & Denial of Service & Tấn công từ chối dịch vụ \\
\textbf{DVWA} & Damn Vulnerable Web Application & Ứng dụng web kiểm thử lỗ hổng an ninh \\
\textbf{Eavesdropping} & Traffic Eavesdropping & Hành vi nghe lén luồng dữ liệu trên đường truyền \\
\textbf{HSTS} & HTTP Strict Transport Security & Cơ chế bảo mật bắt buộc dùng giao thức HTTPS \\
\textbf{HTTP} & Hypertext Transfer Protocol & Giao thức truyền tải siêu văn bản \\
\textbf{HTTPS} & Hypertext Transfer Protocol Secure & Giao thức truyền tải siêu văn bản an toàn (qua TLS) \\
\textbf{ICMP} & Internet Control Message Protocol & Giao thức điều khiển bản tin Internet \\
\textbf{IDS / IPS} & Intrusion Detection / Prevention System & Hệ thống phát hiện / ngăn chặn xâm nhập \\
\textbf{IP} & Internet Protocol & Giao thức liên mạng (tầng 3) \\
\textbf{MAC} & Media Access Control & Địa chỉ vật lý tầng liên kết dữ liệu (tầng 2) \\
\textbf{MitM} & Man-in-the-Middle & Kỹ thuật tấn công kẻ đứng ở giữa \\
\textbf{MTU} & Maximum Transmission Unit & Đơn vị truyền tải tối đa của gói tin \\
\textbf{NIC} & Network Interface Card & Card giao tiếp mạng \\
\textbf{PKI} & Public Key Infrastructure & Hạ tầng khóa công khai \\
\textbf{PoC} & Proof of Concept & Kịch bản chứng minh tính khả thi kỹ thuật \\
\textbf{RFC} & Request for Comments & Tài liệu chuẩn kỹ thuật của IETF \\
\textbf{Rogue AP} & Rogue Access Point & Điểm truy cập không dây giả mạo / trái phép \\
\textbf{SSL} & Secure Sockets Layer & Giao thức bảo mật tầng truyền tải (tiền thân TLS) \\
\textbf{TLS} & Transport Layer Security & Giao thức bảo mật tầng truyền tải \\
\textbf{VLAN} & Virtual Local Area Network & Mạng cục bộ ảo \\
\textbf{VPN} & Virtual Private Network & Mạng riêng ảo mã hóa \\
\bottomrule
\end{xltabular}
